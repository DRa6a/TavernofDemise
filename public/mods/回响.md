## 二、说明

1. 本**异卷**在**原版规则**的基础上添加**若干回响**，**游戏**过程中**参与者**可根据此**异卷**所定义**回响**对**游戏内容**产生**影响**．
2. **回响名称**均来自**小说《十日终焉》**，为**二次创作**，**无侵权行为**．
3. 若对**此异卷**内容有**异议**，可联系**制作组成员**进行修改，我们会在编写**下一版本**时进行**参考**…我们会听劝哒(！

## 三、定义

1. **回响**：一种**信念**所催生的**力量**或是**能力**，**信念**愈强**力量**愈强，**回响**愈**稳定**，但在此**异卷**中，设定好了**回响**的**使用次数**和**效果**，虽此，仍望各位**参与者们 余念安定**．
2. **回响**能在**特定**的**时间**或**回合**内**干预游戏的进行**．
3. **回响**效果**未消失**时，**不可丢弃**本**回响**．

---

## 四、新添机制

1. **激发回合**：在**选举回合**前添加**激发回合**，每名**参与者**在**回响堆**中抽取 **3** 个**回响**，若对其中某个回响不满意，可**丢弃** 3 个中的 **1** 个重新抽，但也只能替换掉 **1** 个。
2. **自我质疑跳过**：当**开牌回合**内出现**自身必须质疑自身**的情况（自身出了手牌，其余参与者通过回响使手中没有手牌），此时**跳过本大回合**。

> 上述两条机制在下方 `## 钩子` 章节中以 TypeScript 代码实现。

function __hx(s) {
  var h = 2166136261 >>> 0;
  for (var i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i);
    h = Math.imul(h, 16777619);

}

// 抽取 3 个回响（确定性，从全部回响中抽）
function __drawEchoes(rng) {
  var pool = __ECHO_IDS.slice();
  var picked = [];
  for (var i = 0; i < 3 && pool.length > 0; i++) {
    var idx = Math.floor(rng() * pool.length);
    picked.push(pool.splice(idx, 1)[0]);
  }
  return picked.map(function (id) { return { id: id, remaining: __ECHO_MAX[id] }; });
}

// 找一个未被锁定的状态效果（按 id），由「忘忧」使用
function __findRemovableState(player) {
  if (!player.stateEffectIds) return null;
  for (var i = 0; i < player.stateEffectIds.length; i++) {
    var id = player.stateEffectIds[i];
    if (id === 'suoding' || id === 'eyun') continue; // 「锁定」/「厄运」不可被「忘忧」解除
    return id;
  }
  return null;
}

// 游戏开始前：清空每个玩家身上的状态与回响库存
onBeforeGameStart(state) {
  state.players.forEach(function (p) {
    p.stateEffectIds = [];
    p.modData = {
      echoes: [],
      stateDurations: {},
      inspireReplaced: false

// 激发回合：每位玩家抽取 3 个回响。
// 由于「激发回合」是声明式阶段、实际阻塞在 UI 层实现，
// 这里在 onBeforeElection（RoundManager 选举首位玩家前）派发回响。
onBeforeElection(state) {
  var rng = __mkRng(__hx('huixiang:inspire:' + (state.currentRound || 0)));
  state.players.forEach(function (p) {
    if (p.isDead) return;
    __initPlayerModData(p);
    p.modData.echoes = __drawEchoes(rng);
  });
}

// 大回合开始：清理「不灭」以外的死亡标记、递减所有状态持续时间
onBigRoundStart(state, round) {
  state.players.forEach(function (p) {
    if (p.isDead) return;
    __initPlayerModData(p);
    var dur = p.modData.stateDurations;
    Object.keys(dur).forEach(function (sid) {
      dur[sid] -= 1;
      if (dur[sid] <= 0) {
        delete dur[sid];
        p.stateEffectIds = (p.stateEffectIds || []).filter(function (x) { return x !== sid; });
      }
    });
  });
}

// 出牌前：检查「哀昏」状态
onBeforePlay(state, player, cardIds) {
  if ((player.stateEffectIds || []).indexOf('aihun') >= 0) {
    // 跳过本回合：清掉一次性状态、跳过出牌
    player.stateEffectIds = player.stateEffectIds.filter(function (s) { return s !== 'aihun'; });
    if (player.modData && player.modData.stateDurations) delete player.modData.stateDurations['aihun'];
    // 仍走完流程：抛错让调用方处理（实际 UI 应读取此状态后再决定）
    // 这里只记录 modData 标志，不真正抛错
    if (!player.modData) player.modData = {};
    player.modData.skippedByAihun = true;
  }
}

// 开牌前：检测「自我质疑」特殊情形——
// 出牌者上下家均为自己（即只剩自己一手有牌、别人都被「赤炎/魂迁/隐匿」清空），
// 此时本大回合没有意义，直接跳过。
onBeforeOpen(state) {
  var last = state.lastPlay;
  if (!last) return;
  // 上下家寻找
  var order = state.players.filter(function (p) { return !p.isDead; });
  if (order.length < 2) return;
  var idx = -1;
  for (var i = 0; i < order.length; i++) if (order[i].id === state.activePlayerId) { idx = i; break; }
  if (idx < 0) return;
  var prev = order[(idx - 1 + order.length) % order.length];
  var next = order[(idx + 1) % order.length];
  // 若 prev === next === active，则为「自身上下家均为自己」——只剩自己一手有牌
  if (prev && next && prev.id === state.activePlayerId && next.id === state.activePlayerId) {
    if (!state.modData) state.modData = {};
    state.modData.skipBigRound = true;
  }
}

// 开牌后：「招灾」「受控」「失明」「梦迷」等状态应用
onAfterOpen(state, isFake) {
  if (state.modData && state.modData.skipBigRound) {
    // 规则 §4.2：跳过本大回合
    delete state.modData.skipBigRound;
    state.players.forEach(function (p) {
      if (p.isDead) return;
      p.isOutOfRound = true;
    });
  }
}

// 生死回合前：「天行健」可挡住一次天龙制裁。
// 这里仅暴露 modData 标志：若玩家 modData.blockTianlong === true，则强制让 finalFace 不是天龙。
onBeforeLifeDeath(state, loser) {
  __initPlayerModData(loser);
  if (loser.modData.blockTianlong && loser.availableBeasts.indexOf('天龙') >= 0) {
    var faces = loser.availableBeasts.filter(function (f) { return f !== '天龙'; });
    if (faces.length > 0) {
      var rng = __mkRng(__hx('tianlong:' + loser.id + ':' + state.currentRound));
      loser.modData.__forcedFace = faces[Math.floor(rng() * faces.length)];
    }
    loser.modData.blockTianlong = false; // 一次性
  }
}

// 有人死亡时：「不灭」发动——附加「末命」状态，让该玩家在 2 个大回合内不会真正出局
onPlayerDied(state, player) {
  __initPlayerModData(player);
  if (player.modData.echoes.some(function (e) { return e.id === 'bumie' && e.remaining > 0; })) {
    // 消耗「不灭」
    var e = player.modData.echoes.find(function (x) { return x.id === 'bumie'; });
    e.remaining -= 1;
    if (e.remaining <= 0) {
      player.modData.echoes = player.modData.echoes.filter(function (x) { return x.id !== 'bumie'; });
    }
    // 复活 + 挂「末命」
    player.isDead = false;
    player.stateEffectIds = (player.stateEffectIds || []).concat(['momming']);
    player.modData.stateDurations['momming'] = 2;
  }
}

// 大回合结束：清理「末命」触发的复活玩家
onBigRoundEnd(state, round) {
  state.players.forEach(function (p) {
    if (!p.isDead && p.modData && p.modData.stateDurations && p.modData.stateDurations['momming'] === 0) {
      // 「末命」到期后真正死亡
      p.isDead = true;
    }
  });
}

// 给玩家尝试使用某个回响：返回 { ok, reason }，由 UI 根据 reason 决定弹什么提示
function useEcho(player, echoId) {
  if (!player.modData || !player.modData.echoes) return { ok: false, reason: 'no_echoes' };
  var e = player.modData.echoes.find(function (x) { return x.id === echoId; });
  if (!e) return { ok: false, reason: 'not_owned' };
  if (e.remaining <= 0) return { ok: false, reason: 'no_uses_left' };
  e.remaining -= 1;
  if (e.remaining <= 0) {
    player.modData.echoes = player.modData.echoes.filter(function (x) { return x.id !== echoId; });
  }
  return { ok: true };
}
